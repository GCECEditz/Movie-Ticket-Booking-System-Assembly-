;since we're using a SMALL model, we only get 1 Code Segment and 1 Data Segment
;due to SMALL being declared, the user-defined procedure will implicitly be set to "near"
;which brings the same effect of PROC_NAME PROC NEAR
.MODEL	SMALL

;to note, since stack is 64 bytes, the program should not exceed more than (stack_size / 2)
;NEAR procedure calls, as their return pointers take up to 2 bytes each
.STACK	256

		;user-defined structures

		;TICKET, 1 byte for movie, date, time, adult count and children count
		;        6 bytes for seats, initialised to 21 (Out of bounds)
		;        1 + 1 + 1 + 1 + 1 + (6 * 1) = 11 bytes
		TICKET	struc
			MOVIE	DB	0
			DATE	DB	0
			TIME	DB	0
			ADULTS	DB	0
			CHILDS	DB	0
			SEATS	DB	6 DUP (0H)
		TICKET	ends

		;USER, 8 + 1 bytes for username
		;      8 + 1 bytes for password
		;      6 * TICKET (11) bytes for tickets
		;      9 + 9 + (6 * 11) = 84 bytes
		USER	struc
			USERNAME	DB	9 DUP ('$')
			PASSWORD	DB	9 DUP ('$')
			TICKETS		TICKET	6 DUP	(<>)
		USER	ends

;User-defined Procedures

	;the first menu
	START_MENU	PROC
		_START_MENU_BEGIN:
		;clear background
		mov		SCREEN_ATTR, BACKGROUND + GREY
		call	CLEAR_SCREEN

		;fill blue text
		mov		SCREEN_ATTR, BACKGROUND + LIGHT_BLUE
		mov		FILL_START, 0000H
		mov		FILL_END, 114FH
		call	FILL_SCREEN

		;fill grey text
		mov		SCREEN_ATTR, BACKGROUND + GREY
		mov		FILL_START, 1200H
		mov		FILL_END, 184FH
		call	FILL_SCREEN

		;move back the cursor
		mov		CURSOR_POS, 0000H
		call	MOVE_CURSOR

		;display logo
		mov		AH, 09H
		lea		DX, START_MENU_TEXT
		int		21H

		;checks whether user count is full and program should allow another sign-up
		call	COUNT_EXISTING_USERS
		cmp		EXISTING_USERS, 4
		jge		SM_NO_SIGN_UP
		jmp		SM_CAN_SIGN_UP

		SM_NO_SIGN_UP:
		;displays text regarding no sign-up option
		;set option count to 0-1
		mov		CURSOR_POS, 1708H
		call	MOVE_CURSOR
		mov		AH, 09H
		lea		DX, START_MENU_NOSIG_UP_OPT
		int		21H

		mov		MAX_OPTS, 1
		jmp		SM_CHECK_OPTIONS
		
		SM_CAN_SIGN_UP:
		;displays text allowing for sign-up option
		;set option count to 0-2
		mov		CURSOR_POS, 1508H
		call	MOVE_CURSOR
		mov		AH, 09H
		lea		DX, START_MENU_SIGN_UP_MSG
		int		21H
		mov		CURSOR_POS, 1708H
		call	MOVE_CURSOR
		mov		AH, 09H
		lea		DX, START_MENU_SIGN_UP_OPT
		int		21H

		mov		MAX_OPTS, 2
		jmp		SM_CHECK_OPTIONS

		SM_CHECK_OPTIONS:
		;get user's option to determine what menu to open
		mov		CURSOR_POS, 171DH
		call	MOVE_CURSOR

		call	GET_OPTION2
		cmp		FINAL_OPTION, 0
		je		_START_MENU_END
		cmp		FINAL_OPTION, 1
		je		_START_MENU_LOGIN
		cmp		FINAL_OPTION, 2
		je		_START_MENU_SIGN_UP

		;user chose to login, get ready for login, but in case the user entered 0 in the login page
		;to return to this page, jump back to the beginning of this page
		_START_MENU_LOGIN:
		call	LOGIN_MENU
		cmp		RETURN_TO_START, 1
		je		_START_MENU_BEGIN
		call	SW_MAIN_MENU
		jmp		_START_MENU_BEGIN
		
		;user chose to sign up, get ready for sign up, but in case the user entered 0 in the 
		;sign up page to return to this page, jump back to the beginning of this page
		_START_MENU_SIGN_UP:
		call	SIGN_UP_MENU
		cmp		RETURN_TO_START, 1
		je		_START_MENU_BEGIN
		call	SW_MAIN_MENU
		jmp		_START_MENU_BEGIN

		mov		CURSOR_POS, 1708H
		call	MOVE_CURSOR
		mov		_CL_INP_FIELD_LENGTH, 70
		call	_CLEAR_INPUT_FIELD
		;ask if user is sure to quit
		mov		AH, 09H
		lea		DX, QUIT_MSG
		int		21H
		
		mov		CURSOR_POS, 171DH
		call	MOVE_CURSOR
		call	GET_YN
		cmp		YN_RESPONSE, 0
		je		_START_MENU_BEGIN
		;move the cursor to the specified position
		mov		CURSOR_POS, 1800H
		call	MOVE_CURSOR
		ret

	;the login menu
	LOGIN_MENU	PROC
		;clear background
		mov		RETURN_TO_START, 0
		mov		AL, BACKGROUND + 8
		mov		SCREEN_ATTR, AL
		call	CLEAR_SCREEN

		;fill blue text
		mov		AL, BACKGROUND + LIGHT_BLUE
		mov		SCREEN_ATTR, AL
		mov		FILL_START, 0000H
		mov		FILL_END, 064FH

		;fill grey text
		mov		AL, BACKGROUND + GREY
		mov		SCREEN_ATTR, AL
		mov		FILL_START, 0700H
		mov		FILL_END, 184FH
		call	FILL_SCREEN
		
		mov		CURSOR_POS, 0000H
		call	MOVE_CURSOR

		;display login menu text
		mov		AH, 09H
		lea		DX, LOGIN_MENU_TEXT
		int		21H
		
		;getting the username, echo is set to 1 because we want to echo the user input
		LM_GET_USERNAME:
		mov		CURSOR_POS, 0C21H
		call	MOVE_CURSOR
		mov		_CL_INP_FIELD_LENGTH, 8
		call	_CLEAR_INPUT_FIELD
		mov		DO_ECHO, 1
		call	GET_STRING2
		
		cmp		GS_PRESS_BACK, 1
		jne		LM_CONTINUE_USERNAME
		mov		RETURN_TO_START, 1
		ret
		
		LM_CONTINUE_USERNAME:
		;am aware that a user that types in 'NAME$' will get accepted if the username
		;database contains 'NAME$$$$' (trailing $ are not checked)
		mov		DEST_FIELD_OFFSET, OFFSET TO_TEST_USERNAME
		mov		INP_FIELD_OFFSET, OFFSET UNBOUND_STR_ARR
		mov		ACT_FIELD_LEN, 8
		call	COPY_TEXT
		call	CHECK_USER_EXISTS
		
		;check if user exists, then we get password, else we reprompt
		cmp		USER_EXISTS, 1
		je		LM_GET_PASSWORD
		
		;no such user, clear input field
		mov		_CL_INP_FIELD_LENGTH, 8
		call	_CLEAR_INPUT_FIELD

		;clear error message
		mov		_ERROR_MSG_LEN, _NOTCONT_INPUT_LEN
		mov		AL, MAX_UNB_STR_LEN
		mov		_INP_FIELD_LEN, AL
		call	_CLEAR_ERROR_MSG

		;display error message
		mov		_ERROR_MSG_OFFSET, OFFSET USERNAME_MISMATCH_MSG
		mov		AL, MAX_UNB_STR_LEN
		mov		_ERROR_MSG_POS_OFFSET, AL
		call	_DISPLAY_ERROR_MSG
		jmp		LM_GET_USERNAME

		;getting the password, echo is set to 0 because we want to hide the user input
		LM_GET_PASSWORD:
		mov		CURSOR_POS, 0D21H
		call	MOVE_CURSOR
		mov		_CL_INP_FIELD_LENGTH, 8
		call	_CLEAR_INPUT_FIELD
		mov		DO_ECHO, 0
		call	GET_STRING2

		cmp		GS_PRESS_BACK, 1
		jne		LM_CONTINUE_PASSWORD

		;clear input field if user press 0 to reenter username
		mov		_ERROR_MSG_LEN, _NOTCONT_INPUT_LEN
		mov		AL, MAX_UNB_STR_LEN
		mov		_INP_FIELD_LEN, AL
		call	_CLEAR_ERROR_MSG
		mov		_CL_INP_FIELD_LENGTH, 8
		call	_CLEAR_INPUT_FIELD
		jmp		LM_GET_USERNAME

		LM_CONTINUE_PASSWORD:
		;check if password is correct
		mov		SI, CURRENT_USER_OFFSET
		lea		BX, ALL_USERS[SI].PASSWORD
		mov		DEST_FIELD_OFFSET, BX
		mov		INP_FIELD_OFFSET, OFFSET UNBOUND_STR_ARR
		mov		ACT_FIELD_LEN, 8
		call	IS_STRING_SAME

		cmp		RES_IS_SAME, 1
		;password is correct, succesfully log in
		je		LM_DONE
		
		;else, reprompt, clear input field
		mov		_CL_INP_FIELD_LENGTH, 8
		call	_CLEAR_INPUT_FIELD

		;clear error message
		mov		_ERROR_MSG_LEN, _NOTCONT_INPUT_LEN
		mov		AL, MAX_UNB_STR_LEN
		mov		_INP_FIELD_LEN, AL
		call	_CLEAR_ERROR_MSG

		;display error message
		mov		_ERROR_MSG_OFFSET, OFFSET PASSWORD_MISMATCH_MSG
		mov		AL, MAX_UNB_STR_LEN
		mov		_ERROR_MSG_POS_OFFSET, AL
		call	_DISPLAY_ERROR_MSG
		jmp		LM_GET_PASSWORD

		LM_DONE:
		ret

	;buy seats menu

		;clear screen
		mov		SCREEN_ATTR, BACKGROUND + GREY
		call	CLEAR_SCREEN
		call	DISPLAY_SEAT_LAYOUT

		BS_ADULT:
		;get the adult pax
		mov		CURSOR_POS, 0B00H
		call	MOVE_CURSOR

		;adult pax text
		mov		AH, 09H
		lea 	DX, PAX_SEL_TEXT
		int		21H

		mov		CURSOR_POS, 0B08H
		call	MOVE_CURSOR
		movzx	AX, REMAINING_SEATS
		mov		HEX_NUM, AX
		call	PRINT_DECIMAL

		;math to find minimum seats left (due to room may have less than 6 seats before selecting)
		mov		AL, REMAINING_SEATS
		mov		MIN_1, AL
		mov		MIN_2, 6
		call	GET_MINIMUM
		mov		AL, MINIMUM
		mov		MAX_OPTS, AL

		;print remaining seats left
		mov		CURSOR_POS, 101DH
		call	MOVE_CURSOR
		movzx	AX, MINIMUM
		mov		HEX_NUM, AX
		call	PRINT_DECIMAL

		;get user selected option
		mov		CURSOR_POS, 1021H
		call	MOVE_CURSOR
		call	GET_OPTION2
		mov		AL, FINAL_OPTION
		mov		SELECTED_ADULT_PAX, AL

		BS_CHILD:
		mov		CURSOR_POS, 0B08H
		call	MOVE_CURSOR

		;child pax text
		mov		AH, 09H
		lea 	DX, EMPTY
		int		21H
		
		;calculate remaining seats after subtracting adult pax
		mov		CURSOR_POS, 0B08H
		call	MOVE_CURSOR
		movzx	AX, REMAINING_SEATS
		sub		AL, SELECTED_ADULT_PAX
		mov		HEX_NUM, AX
		call	PRINT_DECIMAL

		;math to find minimum seats left (due to room may have less than 6 seats before selecting)
		;while taking into account the adult seats
		mov		AL, REMAINING_SEATS
		sub		AL, SELECTED_ADULT_PAX
		mov		MIN_1, AL
		mov		AL, 6
		sub		AL, SELECTED_ADULT_PAX
		mov		MIN_2, AL
		call	GET_MINIMUM
		mov		AL, MINIMUM
		mov		MAX_OPTS, AL

		;print remaining seats left
		mov		CURSOR_POS, 111DH
		call	MOVE_CURSOR
		movzx	AX, MINIMUM
		mov		HEX_NUM, AX
		call	PRINT_DECIMAL

		;get user selected option
		mov		CURSOR_POS, 1121H
		call	MOVE_CURSOR
		call	GET_OPTION2
		mov		AL, FINAL_OPTION
		mov		SELECTED_CHILD_PAX, AL
		ret

	;calculate subtotal (multiplied by 100)
	CALCULATE_SUBTOTAL	PROC
		 ; SUBTOTAL = ADULTS*15 + CHILDREN*8  (store in cents: x100)
		movzx 	AX, SELECTED_ADULT_PAX
		mov		BX, ADULTPRICE
		mul		BX
		mov		BX, 100
		mul		BX
		mov		SUBTOTAL, AX

		movzx	AX, SELECTED_CHILD_PAX
		mov		BX, CHILDPRICE
		mul		BX
		mov		BX, 100
		mul		BX
		add		SUBTOTAL, AX
		ret
	CALCULATE_SUBTOTAL	ENDP
	
	;calculate discount (multiplied by 100)
	CALCULATE_DISCOUNT	PROC
		; --- DISCOUNT CALC ---
		mov		AX, SUBTOTAL
		xor		DX, DX
		mov		BX, 2000
		div		BX
		mov		CX, AX

		;calculate the number of 5% discount blocks
		mov		AX, SUBTOTAL
		mov		BX, 5
		mul		BX
		mov		BX, 100
		div		BX
		mov		BX, CX
		mul		BX
		mov		DISCOUNT, AX
		ret
	CALCULATE_DISCOUNT	ENDP

	;calculate final price (multiplied by 100)
	CALCULATE_TOTAL	PROC
		; FINALPRICE = SUBTOTAL - DISCOUNT
		mov		AX, SUBTOTAL
		sub		AX, DISCOUNT
		mov		FINALPRICE, AX
		ret
	CALCULATE_TOTAL	ENDP

	;transfer seats from SELECTED_SEATS to ALL_SEATS with the correct offset
	TRANSFER_SEATS	PROC
		;loop 6 times (max 6 seats)
		mov		CX, 6
		mov		SI, 0
		TS_SEAT:
		
		;check if seat id is not default (invalid seat, as user did not choose it and is
		;initalised to 0 at first)
		mov		BL, SELECTED_SEATS[SI]
		cmp		BL, 0
		je		TS_CONTINUE_LOOP
		mov		SELECTED_SEAT_IDX, BL
		
		;get seat offset to replace the selected seat from 'O' to 'X'
		call	GET_SEAT_OFFSET
		mov		DI, SEAT_OFFSET
		mov		ALL_SEATS[DI], 'X'
		
		;continue loop, add offset
		TS_CONTINUE_LOOP:
		inc		SI
		loop	TS_SEAT
		ret
	TRANSFER_SEATS	ENDP

		;transfer child data
		mov		AL, SELECTED_CHILD_PAX
		mov		[BX], AL
		inc		BX

		;transfer selected seats data by copying the 6 bytes
		mov		DEST_FIELD_OFFSET, BX
		mov		INP_FIELD_OFFSET, OFFSET SELECTED_SEATS
		mov		ACT_FIELD_LEN, 6
		call	COPY_TEXT

		ret
	ADD_TICKET_DATA_TO_USER	ENDP

	;remove existing ticket seat details from ALL_SEATS 
	REMOVE_SEATS	PROC
		;loop 6 times (max 6 seats)
		mov		CX, 6
		mov		SI, 0
		RS_SEAT:
		
		;check if seat id is not default (invalid seat, as user did not choose it and is
		;initalised to 0 at first)
		mov		BL, SELECTED_SEATS[SI]
		cmp		BL, 0
		je		RS_CONTINUE_LOOP
		mov		SELECTED_SEAT_IDX, BL
		
		;get seat offset to replace the selected seat from 'X' to 'O'
		call	GET_SEAT_OFFSET
		mov		DI, SEAT_OFFSET
		mov		ALL_SEATS[DI], 'O'
		
		;continue loop, add offset
		RS_CONTINUE_LOOP:
		inc		SI
		loop	RS_SEAT
		ret
	REMOVE_SEATS	ENDP

	;remove existing ticket details from ALL_USERS 
	REMOVE_TICKET_DATA_FROM_USER	PROC
		;since all 11 bytes are initalised to 0 for an empty ticket, just loop through
		;11 times resetting all to 0
		mov		CX, SIZE TICKET
		mov		SI, CURRENT_USER_OFFSET
		lea		DI, ALL_USERS[SI].TICKETS
		mov		SI, CURRENT_TICKET_OFFSET
		add		DI, SI
		mov		AL, 0
		
		;loop by replacing the value at [DI] with 0
		RTDFU_LOOP:
		mov		[DI], AL
		inc		DI
		loop	RTDFU_LOOP
		ret
	REMOVE_TICKET_DATA_FROM_USER	ENDP

	;count existing purchased tickets and find EMPTY_TICKET_OFFSET (offset of any empty ticket)
	COUNT_EXISTING_TICKETS	PROC
		;reset EMPTY_TICKET_OFFSET to -1 (not found) and EXISTING_TICKETS to 0
		mov		EMPTY_TICKET_OFFSET, -1
		mov		CX, 6
		mov		EXISTING_TICKETS, 0
		mov		DI, CURRENT_USER_OFFSET
		mov		SI, 0

		;loop through each ticket offset to check if there is data there (not 0)
		CET_TICKET:
		lea		BX, ALL_USERS[DI].TICKETS
		mov		DL,	[BX + SI]
		mov		AL, 0
		cmp		DL, AL
		je		CET_NO_USER
		
		;there is ticket, add EXISTING_TICKETS by 1
		inc		EXISTING_TICKETS
		jmp		CET_CONTINUE
		
		;there is no ticket, set EMPTY_TICKET_OFFSET to the first empty ticket offset
		CET_NO_USER:
		cmp		EMPTY_TICKET_OFFSET, 0
		jge		CET_CONTINUE
		mov		EMPTY_TICKET_OFFSET, SI
		jmp		CET_CONTINUE
		
		;continue loop
		CET_CONTINUE:
		add		SI, SIZE TICKET
		loop	CET_TICKET
		ret
	COUNT_EXISTING_TICKETS	ENDP

	;clear ticket buffer to help reorder ticket data after user deletes one
	CLEAR_TICK_BUFFER	PROC
		;clear all 6 ticket arrays in the buffer to 0 
		mov		CX, 66
		mov		SI, 0
		CTB_CLEAR:
		mov		USER_TICKET_BUFFER[SI], 0
		inc		SI
		loop	CTB_CLEAR
		ret
	CLEAR_TICK_BUFFER	ENDP

	;reorder all tickets so that all tickets are continuous (no empty gaps between tickets)
	;this avoids random displaying bugs when displaying existing tickets
	REORDER_EXISTING_TICKETS	PROC
		;clear the ticket buffer
		call	CLEAR_TICK_BUFFER
		
		;loop 6 times (max 6 tickets)
		mov		CX, 6
		mov		DI, CURRENT_USER_OFFSET
		mov		SI, 0
		lea		BX, ALL_USERS[DI].TICKETS
		mov		DI, OFFSET USER_TICKET_BUFFER

		RET_TICKET:
		;compare if ticket data is empty
		push	CX
		mov		DL,	[BX + SI]
		mov		AL, 0
		cmp		DL, AL
		je		RET_CONTINUE
		
		;else, we copy any entries to the buffer in order
		mov		AX, BX
		add		AX, SI
		mov		INP_FIELD_OFFSET, AX
		mov		DEST_FIELD_OFFSET, DI
		mov		ACT_FIELD_LEN, SIZE TICKET
		push	BX
		push	DI
		push	SI
		call	COPY_TEXT
		pop		SI
		pop		DI
		pop		BX
		
		;then add the offset
		add		DI, SIZE TICKET
		jmp		RET_CONTINUE
		
		RET_CONTINUE:
		;then add the offset
		add		SI, SIZE TICKET
		pop		CX
		loop	RET_TICKET
		
		;after done looping, we copy the buffer back to the offset of the current user to
		;overwrite all ticket data to a reordered version
		mov		DI, CURRENT_USER_OFFSET
		lea		BX, ALL_USERS[DI].TICKETS
		mov		INP_FIELD_OFFSET, OFFSET USER_TICKET_BUFFER
		mov		DEST_FIELD_OFFSET, BX
		mov		ACT_FIELD_LEN, 66
		call	COPY_TEXT
		ret
	REORDER_EXISTING_TICKETS	ENDP

	;show all X and O seats
	SHOW_SEATS	PROC
		;loop 20 times as there are 20 seats in the room
		mov		REMAINING_SEATS, 20
		mov		CURSOR_POS, 051EH
		call	MOVE_CURSOR
		mov		SELECTED_SEAT_IDX, 1

		;2 rows
		mov		CX, 2
		SS_OUTER:
		push	CX
		mov		DX, CURSOR_POS
		mov		CX, 10

		;10 columns
		SS_INNER:
		push	CX
		push	DX
		;check the seat offset in ALL_SEATS to check if seat is taken
		call	GET_SEAT_OFFSET
		pop		DX
		
		mov		SI, SEAT_OFFSET
		mov		AL, ALL_SEATS[SI]
		;taken
		cmp		AL, 'X'
		je		SS_TAKEN
		;vacant
		cmp		AL, 'O'
		je		SS_VACANT
		jmp		SS_CONTINUE_INNER
		
		;if taken, set text to red and print 'X '
		SS_TAKEN:
		mov		FILL_START, DX
		inc		DX
		mov		FILL_END, DX
		mov		SCREEN_ATTR, BACKGROUND + RED
		push	DX
		call	FILL_SCREEN
		mov		AH, 09H
		lea		DX, S_TAKEN
		int		21H
		pop		DX
		dec		REMAINING_SEATS
		jmp		SS_CONTINUE_INNER
		
		;if taken, set text to green and print 'O '
		SS_VACANT:
		mov		FILL_START, DX
		inc		DX
		mov		FILL_END, DX
		mov		SCREEN_ATTR, BACKGROUND + GREEN
		push	DX
		call	FILL_SCREEN
		mov		AH, 09H
		lea		DX, S_VACANT
		int		21H
		pop		DX
		jmp		SS_CONTINUE_INNER

		;then loop and check the next seat of the room
		SS_CONTINUE_INNER:
		inc		DX
		inc		SELECTED_SEAT_IDX
		pop		CX
		loop	SS_INNER
		
		;after the first row, jump to next row
		add		CURSOR_POS, 100H
		call	MOVE_CURSOR
		pop		CX
		loop	SS_OUTER

		ret
	SHOW_SEATS	ENDP

	;shows the selected seats replacing with a blue '?'
	SHOW_SELECTED_SEATS	PROC
		mov		CX, 6
		mov		SI, 0

		;loop through all seats
		SSS_SELECTED:
		push	CX
		mov		AX, 0
		mov		BH, SELECTED_SEATS[SI]
		
		cmp		BH, 0
		ja		SSS_DRAW_SEAT
		jmp		SSS_CONTINUE

		;seat exists, move cursor and fill it to blue and display '?'
		;reposition cursor
		SSS_DRAW_SEAT:
		dec		BH
		mov		AL, BH
		mov		BL, 10
		div		BL
		add		AL, 5H
		mov		DH, AL
		mov		BL, 2
		mov		AL, AH
		xor		AH, AH
		mul		BL
		add		AL, 1EH
		mov		DL, AL

		;set text to blue, and print the ?
		mov		FILL_START, DX
		inc		DX
		mov		FILL_END, DX
		mov		SCREEN_ATTR, BACKGROUND + BLUE
		mov		CURSOR_POS, DX
		call	MOVE_CURSOR
		call	FILL_SCREEN
		mov		AH, 09H
		lea		DX, S_SELECTED
		int		21H

		;seat doesn't exist, continue 
		SSS_CONTINUE:
		inc		SI
		pop		CX
		loop	SSS_SELECTED
		ret
	SHOW_SELECTED_SEATS	ENDP

	;gets the absolute seat offset based on SELECTED_MOVIE_IDX, SELECTED_DATE_IDX, SELECTED_TIME_IDX,
	;SELECTED_SEAT_IDX
	;formula is as of below:
	;--------------------------------------------------------------
	;SELECTED_MOVIE_IDX * (DATE_COUNT * TIME_COUNT * SEAT_COUNT) +
	;SELECTED_DATE_IDX  * (TIME_COUNT * SEAT_COUNT) +
	;SELECTED_TIME_IDX  * (SEAT_COUNT) +
	;SELECTED_SEAT_IDX
	;--------------------------------------------------------------
	GET_SEAT_OFFSET	PROC
		mov		SEAT_OFFSET, 0
		movzx	AX, SELECTED_MOVIE_IDX
		dec		AX
		mov		BX, DATE_COUNT * TIME_COUNT * SEAT_COUNT
		mul		BX
		add		SEAT_OFFSET, AX
		
		movzx	AX, SELECTED_DATE_IDX
		dec		AX
		mov		BX, TIME_COUNT * SEAT_COUNT
		mul		BX
		add		SEAT_OFFSET, AX
		
		movzx	AX, SELECTED_TIME_IDX
		dec		AX
		mov		BX, SEAT_COUNT
		mul		BX
		add		SEAT_OFFSET, AX
		
		movzx	AX, SELECTED_SEAT_IDX
		dec		AX
		add		SEAT_OFFSET, AX
		ret
	GET_SEAT_OFFSET	ENDP

	;read user data from the file
	READ_USERS_FILE	PROC
		mov		FILE_NAME_OFFSET, OFFSET USER_SAVE_FILE
		call	OPEN_FILE

		cmp		FILE_ERROR_CODE, 0
		jne		_RUF_NO_FILE
		jmp		_RUF_READ

		_RUF_NO_FILE:				; ignore if file does not exist
		ret
		
		_RUF_READ:
		mov		TO_READ_SIZE, SIZE ALL_USERS
		mov		TO_READ_BUFFER_OFFSET, OFFSET ALL_USERS
		call	READ_FILE

		call	CLOSE_FILE
		ret
	READ_USERS_FILE	ENDP

	;write user data into the file
	WRITE_USERS_FILE	PROC

		cmp		FILE_ERROR_CODE, 0
		jne		_WUF_CREATE_FILE
		jmp		_WUF_WRITE

		_WUF_CREATE_FILE:
		call	CREATE_FILE
		jmp		_WUF_WRITE
		
		_WUF_WRITE:
		mov		TO_WRITE_SIZE, SIZE ALL_USERS
		mov		TO_WRITE_BUFFER_OFFSET, OFFSET ALL_USERS
		call	WRITE_FILE
		
		call	CLOSE_FILE
		ret
	WRITE_USERS_FILE	ENDP

	;read seat data from the file
	READ_SEATS_FILE	PROC

		cmp		FILE_ERROR_CODE, 0
		jne		_RSF_NO_FILE
		jmp		_RSF_READ

		_RSF_NO_FILE:				; ignore if file does not exist
		ret
		
		_RSF_READ:
		mov		TO_READ_SIZE, SIZE ALL_SEATS
		mov		TO_READ_BUFFER_OFFSET, OFFSET ALL_SEATS
		call	READ_FILE

	READ_SEATS_FILE	ENDP

	;write seat data into the file
	WRITE_SEATS_FILE	PROC
		mov		FILE_NAME_OFFSET, OFFSET SEAT_SAVE_FILE

		cmp		FILE_ERROR_CODE, 0
		jne		_WSF_CREATE_FILE
		jmp		_WSF_WRITE

		_WSF_CREATE_FILE:
		call	CREATE_FILE
		jmp		_WSF_WRITE
		
		_WSF_WRITE:
		mov		TO_WRITE_SIZE, SIZE ALL_SEATS

		call	WRITE_FILE
		
		call	CLOSE_FILE
		ret
	WRITE_SEATS_FILE	ENDP

	;counts the existing users
	;returns EXISTING_USERS (number of existing users)
	COUNT_EXISTING_USERS	PROC
		;set EXISTING_USERS to 0
		mov		EMPTY_USER_OFFSET, -1
		mov		EXISTING_USERS, 0
		mov		CX, 4
		mov		BX, 0
		mov		SI, 0

		;checks if each username for each user starts with '$' (no input record)
		_CEU_CHECK_USER:
		mov		AL, ALL_USERS[SI].USERNAME[0]
		cmp		AL, '$'
		je		_CEU_NO_USER

		;if record found (username exists), increase EXISTING_USERS
		;and add the pointer to get the next user
		inc		EXISTING_USERS
		add		SI,	SIZE USER
		loop	_CEU_CHECK_USER
		ret

		;no record (USERNAME[0] = '$'), just add the pointer to get the next user
		_CEU_NO_USER:
		mov		EMPTY_USER_OFFSET, SI
		add		SI,	SIZE USER
		loop	_CEU_CHECK_USER
		ret
	COUNT_EXISTING_USERS	ENDP

	;checks if the user exists based on the username
	;requires TO_TEST_USERNAME, the username to test
	;returns CURRENT_USER_OFFSET (the offset of the found user (-1 if none))
	CHECK_USER_EXISTS	PROC
		;reset USER_EXISTS to 0 and initialise the loop
		mov		USER_EXISTS, 0
		mov		BX, 0
		mov		CX, 4
		
		;gets ready to call IS_STRING_SAME to compare existing usernames in the database
		;and the one to test
		CUE_CHECK_USER:
		push	CX
		mov		INP_FIELD_OFFSET, OFFSET TO_TEST_USERNAME
		lea		SI, [ALL_USERS + BX].USERNAME
		mov		DEST_FIELD_OFFSET, SI
		mov		ACT_FIELD_LEN, 8
		call	IS_STRING_SAME
		pop		CX
		
		;check if user has been found (both strings match) 
		cmp		RES_IS_SAME, 1
		je		CUE_FOUND_USER

		;we have not found the user, we move on to the next user object by
		;adding the size of a user to BX
		add		BX, SIZE USER
		loop	CUE_CHECK_USER
		
		;everything falls through, that means the user was not found, CURRENT_USER_OFFSET is
		;set to -1 (invalidated)
		mov		CURRENT_USER_OFFSET, -1
		ret

		;has been found, we return CURRENT_USER_OFFSET as the offset of the found USER
		;and set USER_EXISTS to 1
		CUE_FOUND_USER:
		mov		CURRENT_USER_OFFSET, BX
		mov		USER_EXISTS, 1
		ret
	CHECK_USER_EXISTS	ENDP
	
	;clears SELECTED_SEATS
	CLEAR_SEATS	PROC
		mov		CX, 6
		mov		SI, 0
		CS_CLEAR_SEATS:
		mov		SELECTED_SEATS[SI], 0H
		inc		SI
		loop	CS_CLEAR_SEATS
		ret
	CLEAR_SEATS	ENDP

	;convert the seat coordinate to a number, where (first character * 10) + second character
	_CONVERT_TO_SEAT_ID	PROC
		mov		BX, 0
		mov		DL, UNBOUND_STR_ARR[BX]
		mov		DH, UNBOUND_STR_ARR[BX + 1]
		sub		DL, '1'

		;take into account upper and lower case
		cmp		DH, 'a'
		jae		_CTSI_LOWER
		sub		DH, '@'
		jmp		_CTSI_CON

		_CTSI_LOWER:
		sub		DH, '`'
		
		;start conversion
		_CTSI_CON:
		xor		AX, AX
		mov		BL, 10
		mov		AL, DL
		mul		BL
		
		add		AL, DH
		mov		SEAT_ID, AL
		ret
	_CONVERT_TO_SEAT_ID	ENDP

;File related Procedures----------------------------------------------------------------------------

	;creates a file,
	;requires FILE_NAME_OFFSET (the offset of the file name),
	;returns FILE_HANDLE (the file handle)
	CREATE_FILE	PROC
		;AH = 3CH -> create file instruction
		;CX = 0   -> create a read-only file
		;create a file with the file name from offset FILE_NAME_OFFSET

		;let's clear the file error code
		mov		FILE_ERROR_CODE, 0000H

		mov		AH, 3CH
		mov		CX, 0
		mov		DX, FILE_NAME_OFFSET
		int		21H
		jc		_CF_FAIL

		;success, now let's store the file handle onto FILE_HANDLE
		;and print the file creation success message
		mov		FILE_HANDLE, AX
		mov		AH, 09H
		lea		DX, _FILE_CREATE_SUCCESS
		int		21H
		ret

		;file creation has failed (Path not found)
		_CF_FAIL:
		;print the file creation error message
		mov		FILE_ERROR_CODE, AX
		mov		AH, 09H
		lea		DX, _FILE_CREATE_FAIL
		int		21H
		ret
	CREATE_FILE	ENDP

	;opens a file,
	;requires FILE_NAME_OFFSET (the offset of the file name),
	;returns FILE_HANDLE (the file handle)
	OPEN_FILE	PROC
		;AH = 3DH -> open file instruction
		;AL = 02H -> read and write access mode
		;open a file with the file name from offset FILE_NAME_OFFSET

		;let's clear the file error code
		mov		FILE_ERROR_CODE, 0000H

		mov		AH, 3DH
		mov		AL, 02H
		mov		DX, FILE_NAME_OFFSET
		int		21H
		jc		_OF_NO_FILE

		;success, now let's store the file handle onto FILE_HANDLE
		;and print the file opening success message
		mov		FILE_HANDLE, AX
		mov		AH, 09H
		lea		DX, _FILE_OPEN_SUCCESS
		int		21H
		ret

		;file opening has failed (File not found)
		_OF_NO_FILE:
		;print the file opening error message
		mov		FILE_ERROR_CODE, AX
		mov		AH, 09H
		lea		DX, _FILE_OPEN_FAIL
		int		21H
		ret
	OPEN_FILE	ENDP

	;closes a file,
	;requires FILE_HANDLE (the file handle)
	CLOSE_FILE	PROC
		;AH = 3EH -> close file instruction
		;BX = file handle
		;close a file with the file handle FILE_HANDLE

		;let's clear the file error code
		mov		FILE_ERROR_CODE, 0000H

		mov		AH, 3EH
		mov		BX, FILE_HANDLE
		int		21H
		jc		_CF_FILE_CLOSE_FAIL
		
		;success, the file has been closed
		;and print the file closing success message
		mov		AH, 09H
		lea		DX, _FILE_CLOSE_SUCCESS
		int		21H
		ret

		;file closing has failed (Invalid handle)
		_CF_FILE_CLOSE_FAIL:
		;print the file closing error message
		mov		FILE_ERROR_CODE, AX
		mov		AH, 09H
		lea		DX, _FILE_CLOSE_FAIL
		int		21H
		ret
	CLOSE_FILE	ENDP

	;reads a file,
	;requires FILE_HANDLE (the file handle),
	;TO_READ_SIZE (size to read),
	;TO_READ_BUFFER_OFFSET (the offset of the buffer which is to store the read data)
	READ_FILE	PROC
		;AH = 3FH -> read file/device instruction
		;BX = file handle
		;CX = size of buffer to read
		;DX = the offset of the buffer to store the read data
		;reads a file with the file handle FILE_HANDLE

		;let's clear the file error code
		mov		FILE_ERROR_CODE, 0000H
		
		mov		AH, 3FH
		mov		BX, FILE_HANDLE
		mov		CX, TO_READ_SIZE
		mov		DX, TO_READ_BUFFER_OFFSET
		int		21H
		jc		_RF_FAIL

		;success, the file has been read
		;and print the file reading success message
		mov		AH, 09H
		lea		DX, _FILE_READ_SUCCESS
		int		21H
		ret

		;file reading has failed (Invalid handle)
		_RF_FAIL:
		;print the file reading error message
		mov		FILE_ERROR_CODE, AX
		mov		AH, 09H
		lea		DX, _FILE_READ_FAIL
		int		21H
		ret
	READ_FILE	ENDP

	;writes to a file,
	;requires FILE_HANDLE (the file handle),
	;TO_WRITE_SIZE (size to write),
	;TO_WRITE_BUFFER_OFFSET (the offset of the buffer to write from)
	WRITE_FILE	PROC
		;AH = 340H -> write to file/device instruction
		;BX = file handle
		;CX = size of buffer to write
		;DX = the offset of the buffer to write from
		;reads a file with the file handle FILE_HANDLE

		;let's clear the file error code
		mov		FILE_ERROR_CODE, 0000H

		mov		AH, 40H
		mov		BX, FILE_HANDLE
		mov		CX, TO_WRITE_SIZE
		mov		DX, TO_WRITE_BUFFER_OFFSET
		int		21H
		jc		_WF_FAIL

		;success, the file has been written to
		;and print the file writing success message
		mov		AH, 09H
		lea		DX, _FILE_WRITE_SUCCESS
		int		21H
		ret

		;file writing has failed (Invalid handle)
		_WF_FAIL:
		;print the file writing error message
		mov		FILE_ERROR_CODE, AX
		mov		AH, 09H
		lea		DX, _FILE_WRITE_FAIL
		int		21H
		ret
	WRITE_FILE	ENDP

;Terminal related Procedures------------------------------------------------------------------------

	;moves the cursor to the specified position
	;requires CURSOR_POS (YYXX position of the cursor on the screen)
	MOVE_CURSOR	PROC
		;AH = 02H -> move cursor instruction
		;BH = 00H -> page number 0
		;DX = cursor position
		xor		BX, BX
		mov		AH, 02H
		mov		BH, 00H
		mov		DX, CURSOR_POS
		int		10H						; sets the cursor to specified location
		ret
	MOVE_CURSOR	ENDP

	;prints a newline
	PRINT_NEWLINE	PROC
		;AH = 02H -> display character instruction
		;DL = 0DH -> line feed character
		;DL = 0AH -> carriage return
		mov		AH, 02H
		mov		DL, 0DH
		int		21H
		mov		DL,0AH
		int		21H
		ret
	PRINT_NEWLINE	ENDP

	;clears the entire screen with preset variables
	;requires SCREEN_ATTR (screen attributes)
	CLEAR_SCREEN	PROC
		;AH = 06H -> scroll window up instruction
		;AL = 00H -> number of lines to scroll (0, whole window will be blanked)
		;BH = screen attributes, where the last 4 bits (M-S-Byte) are the background colour
		;and the first 4 bits (L-S-Byte) are the text colour
		;CX = 0000H -> starting coordinate
		;DX = 184FH -> end coordinate
		xor		BX, BX
		mov		AX, 0600H
		mov		BH, SCREEN_ATTR
		mov		CX, 0000H
		mov		DX, 184FH
		int		10H
	CLEAR_SCREEN	ENDP

	;fills the screen with preset variables
	;requires SCREEN_ATTR (screen attributes),
	;FILL_START (YYXX),
	;FILL_END (YYXX)
	FILL_SCREEN	PROC
		;AH = 02H -> move cursor instruction
		;BH = 00H -> page number 0
		;DX = cursor position
		xor		BX, BX
		mov		AH, 02H
		mov		BH, 00H
		mov		DX, FILL_START
		int		10H
		
		;AH = 06H -> scroll window up instruction
		;AL = 00H -> number of lines to scroll (0, whole window will be blanked)
		;BH = screen attributes, where the last 4 bits (M-S-Byte) are the background colour
		;and the first 4 bits (L-S-Byte) are the text colour
		;CX = FILL_START -> starting coordinate
		;DX = FILL_END -> end coordinate
		xor		BX, BX
		mov		AX, 0600H
		mov		BH, SCREEN_ATTR
		mov		CX, FILL_START
		mov		DX, FILL_END
		int		10H
		ret
	FILL_SCREEN	ENDP

;Get input Procedures-------------------------------------------------------------------------------

	;checks if the input is numeric
	;requires IIN_INP_ARR_LEN (length of the buffer to check),
	;IIN_INP_ARR_OFF (offset of the buffer to check)
	;returns RES_IS_NUM (result is number?)
	IS_INPUT_NUMERIC	PROC
		;resets RES_IS_NUM back to false
		mov		RES_IS_NUM, 0
		cmp		IIN_INP_ARR_LEN, 0
		ja		_IIN_START_SCAN_ARR
		
		;input is empty, is not numeric
		ret

		;start scanning the array
		;moves the offset of the character array to SI
		;moves IIN_INP_ARR_LEN to _IIN_INP_AR_IDX for index counting
		_IIN_START_SCAN_ARR:
		movzx	CX, IIN_INP_ARR_LEN
		mov		SI, IIN_INP_ARR_OFF

		;check if the ascii equivalent is out of bounds
		_IIN_SCAN_ARR:
		mov		AL, 57
		cmp		[SI], AL				; more than "9"
		ja		_IIN_IS_NOT_NUMERIC
		mov		AL, 48
		cmp		[SI], AL				; less than "0"
		jb		_IIN_IS_NOT_NUMERIC

		;no issues, continue scanning and increase the memory pointer
		inc		SI

		;index is still > 0, continue scanning
		loop	_IIN_SCAN_ARR
		
		;conclude that character array is indeed numeric, since all checks pass
		jmp		_IIN_IS_NUMERIC

		;any jumps here indicates that at least one character is not (0-9)
		;as long as one of the elements are out of bounds, the input is not numeric anymore
		_IIN_IS_NOT_NUMERIC:
		ret

		;if we have exhausted all elements, without jumping to
		;_IIN_IS_NOT_NUMERIC, it indicates that all characters are numeric
		;RES_IS_NUM is set to 1 (true)
		_IIN_IS_NUMERIC:
		mov		RES_IS_NUM, 1
		ret
	IS_INPUT_NUMERIC	ENDP

	;checks if the input does not contain space " "
	;requires IIC_INP_ARR_LEN (length of the buffer to check),
	;IIC_INP_ARR_OFF (offset of the buffer to check)
	;returns RES_IS_CONT (result is continuous?)
	IS_INPUT_CONTINUOUS	PROC
		;resets RES_IS_CONT back to false
		;resets whether the checker has encountered space for the first time
		mov		_IIC_HAS_ENCOUNTER_SPC, 0
		mov		RES_IS_CONT, 1
		cmp		IIC_INP_ARR_LEN, 0
		ja		_IIC_START_SCAN_ARR
		
		;input is empty, is not continuous
		ret

		;start scanning the array
		;moves the offset of the character array to SI
		;moves IIC_INP_ARR_LEN to _IIC_INP_AR_IDX for index counting
		_IIC_START_SCAN_ARR:
		movzx	CX, IIC_INP_ARR_LEN
		mov		SI, IIC_INP_ARR_OFF

		;check if the ascii equivalent is space (32 / 20H)
		_IIC_SCAN_ARR:
		mov		AL, 32
		cmp		[SI], AL
		je		_IIC_ENCOUNTERED_SPACE

		;if we managed to fall through here, it means the checker has
		;encountered space and the next character is not space
		cmp		_IIC_HAS_ENCOUNTER_SPC, 1
		je		_IIC_IS_NOT_CONT
		
		;no issues, we can continue scanning
		jmp		_IIC_CONTINUE_SCAN_ARR

		;checker has encountered space, but we continue in case
		;the space is at the beginning or at the end
		_IIC_ENCOUNTERED_SPACE:
		mov		_IIC_HAS_ENCOUNTER_SPC, 1
		jmp		_IIC_CONTINUE_SCAN_ARR

		;increase the memory pointer
		_IIC_CONTINUE_SCAN_ARR:
		inc		SI

		;index is still > 0, continue scanning
		loop	_IIC_SCAN_ARR

		;conclude that character array is indeed continuous, since all checks pass
		jmp		_IIC_IS_CONT

		;any jumps here indicates that at least two spaces that are not
		;consecutive have been found by the checker
		;therefore it is not continuous
		_IIC_IS_NOT_CONT:
		mov		RES_IS_CONT, 0
		ret

		;if we have exhausted all elements, without jumping to
		;_IIC_IS_NOT_CONT, it indicates that the buffer is continuous
		;RES_IS_CONT is set to 1 (true)
		_IIC_IS_CONT:
		ret
	IS_INPUT_CONTINUOUS	ENDP

	;compares between two strings to check if their contnts are the same
	;requires ACT_FIELD_LEN (the length to check),
	;INP_FIELD_OFFSET (the offset of the first buffer to check)
	;DEST_FIELD_OFFSET (the offset of the the second buffer to check)
	;returns RES_IS_SAME (result is same?)
	IS_STRING_SAME	PROC
		;compare the length to check
		mov		RES_IS_SAME, 1
		cmp		ACT_FIELD_LEN, 0
		ja		_ISS_START_COMPARE
		
		;length is 0, doesn't require checking
		ret

		_ISS_START_COMPARE:
		;loads the input buffer to SI and the destination buffer to DI
		;sets CX to the length to check
		mov		CX, ACT_FIELD_LEN
		mov		SI, INP_FIELD_OFFSET
		mov		DI, DEST_FIELD_OFFSET
		jmp		_IIS_CONTINUE_COMPARE

		;CX > 0, we continue checking
		_IIS_CONTINUE_COMPARE:
		;transfers the content of [SI] to AL and [DI] to AH
		;increases the index by 1
		mov		AL, [SI]
		mov		AH, [DI]
		cmp		AL, AH
		jne		_IIS_NOT_SAME
		inc		SI
		inc		DI
		loop	_IIS_CONTINUE_COMPARE

		;text has been copied
		ret
		
		;text is not the same, update RES_IS_SAME to false
		_IIS_NOT_SAME:
		mov		RES_IS_SAME, 0
		ret
	IS_STRING_SAME	ENDP

	;checks if a byte value is in an array
	;requires ACT_FIELD_LEN (the length to check),
	;INP_FIELD_OFFSET (the offset of the first buffer to check),
	;TO_FIND (the byte item to find)
	;returns ITEM_FOUND (1 if found, 0 if not)
	FIND_IN	PROC
		;load item to find and reset ITEM_FOUND
		mov		AL, TO_FIND
		mov		ITEM_FOUND, 0
		mov		CX, ACT_FIELD_LEN
		mov		SI, INP_FIELD_OFFSET
		
		;compare AL with every item in the array
		FI_CHECK_ITEM:
		cmp		[SI], AL
		je		FI_FOUND_ITEM
		inc		SI
		loop	FI_CHECK_ITEM
		ret
		
		;if found, set ITEM_FOUND to 1
		FI_FOUND_ITEM:
		mov		ITEM_FOUND, 1
		ret
	FIND_IN	ENDP

	;clears error messages
	_CLEAR_ERROR_MSG	PROC
		;set the area of the error message to red
		mov		AL, BACKGROUND
		add		AL, RED
		mov		SCREEN_ATTR, AL

		movzx	AX, _INP_FIELD_LEN		; set fill end
		add		AX, CURSOR_POS
		add		AX, 1
		movzx	BX, _ERROR_MSG_LEN
		add		BX, AX
		mov		FILL_END, BX			; set fill end coord
		mov		FILL_START, AX			; set fill start coord
		call	FILL_SCREEN				; fill screen

		xor		BX, BX
		mov		AH, 02H
		mov		BH, 00
		mov		DX, FILL_START
		int		10H
		ret
	_CLEAR_ERROR_MSG	ENDP

	;displays error messages
	_DISPLAY_ERROR_MSG	PROC
		;given their offset the error message will be repositioned right after the input field
		xor		BX, BX
		mov		AH, 02H
		mov		BH, 00
		movzx	DX, _ERROR_MSG_POS_OFFSET
		add		DX, CURSOR_POS
		inc		DX
		int		10H
		
		mov		AH, 09H
		mov		DX, _ERROR_MSG_OFFSET
		int		21H

		mov		AH, 02H
		mov		BH, 00
		mov		DX, CURSOR_POS
		int		10H
		ret
	_DISPLAY_ERROR_MSG	ENDP

	;clear the input line
	_CLEAR_INPUT_FIELD	PROC
		;clears by looping ' ' through the input field to 'erase' the input
		xor		BX, BX
		mov		AH, 02H
		mov		BH, 00
		mov		DX, CURSOR_POS
		int		10H

		mov		CX, _CL_INP_FIELD_LENGTH
		mov		AH, 02H
		mov		DL, ' '
		_CIF_CLEAR_LOOP:
		int		21H
		loop	_CIF_CLEAR_LOOP

		mov		AH, 02H
		mov		BH, 00
		mov		DX, CURSOR_POS
		int		10H
		ret
	_CLEAR_INPUT_FIELD	ENDP

	;gets a 1 digit number for option selection
	;requires MAX_OPTS (0 - MAX_OPTS)
	;returns FINAL_OPTION (the final input)
	GET_OPTION2	PROC
		mov		MAX_UNB_STR_LEN, 1
		mov		DO_ECHO, 1
		_GO2_ATTEMPT_OPTION_INPUT:
		call	GET_UNBOUND_STRING

		;loads all required variables before starting the numeric check
		mov		AL, _ACT_UNB_STR_LEN
		mov		IIN_INP_ARR_LEN, AL
		mov		IIN_INP_ARR_OFF, OFFSET UNBOUND_STR_ARR
		call	IS_INPUT_NUMERIC
		cmp		RES_IS_NUM, 0
		je		_GO2_REATTEMPT_OPTION_INPUT_NAN

		;we convert to hex since all characters are digits
		mov		AL, _ACT_UNB_STR_LEN
		mov		_ACT_FIELD_LEN, AL
		mov		_FIELD_OFFSET, OFFSET UNBOUND_STR_ARR
		call	CONVERT_TO_HEX

		;check if the option is with the range
		mov		AX,	DEC_NUM
		cmp		AL, MAX_OPTS
		jg		_GO2_REATTEMPT_OPTION_INPUT_OOR
		jmp		_GO2_STOP_OPTION_INPUT

		_GO2_REATTEMPT_OPTION_INPUT_NAN:
		;clear input field
		mov		_CL_INP_FIELD_LENGTH, 2
		call	_CLEAR_INPUT_FIELD
		
		;clear the error message
		mov		_ERROR_MSG_LEN, _INV_OPT_NAN_LEN
		mov		AL, MAX_UNB_STR_LEN
		mov		_INP_FIELD_LEN, AL
		call	_CLEAR_ERROR_MSG

		;display error msg
		mov		_ERROR_MSG_OFFSET, OFFSET _INV_OPT_NAN_MSG
		mov		AL, MAX_UNB_STR_LEN
		mov		_ERROR_MSG_POS_OFFSET, AL
		call	_DISPLAY_ERROR_MSG
		jmp		_GO2_ATTEMPT_OPTION_INPUT

		_GO2_REATTEMPT_OPTION_INPUT_OOR:
		;clear input field
		mov		_CL_INP_FIELD_LENGTH, 2
		call	_CLEAR_INPUT_FIELD
		;clear the error message
		mov		_ERROR_MSG_LEN, _INV_OPT_NAN_LEN
		mov		AL, MAX_UNB_STR_LEN
		mov		_INP_FIELD_LEN, AL
		call	_CLEAR_ERROR_MSG

		;display error msg
		mov		_ERROR_MSG_OFFSET, OFFSET _INV_OPT_OOR_MSG
		mov		AL, MAX_UNB_STR_LEN
		mov		_ERROR_MSG_POS_OFFSET, AL
		call	_DISPLAY_ERROR_MSG
		jmp		_GO2_ATTEMPT_OPTION_INPUT

		_GO2_STOP_OPTION_INPUT:
		;clear the error message anyway
		mov		_ERROR_MSG_LEN, _INV_OPT_NAN_LEN
		mov		AL, MAX_UNB_STR_LEN
		mov		_INP_FIELD_LEN, AL
		call	_CLEAR_ERROR_MSG

		mov		AX, DEC_NUM
		mov		FINAL_OPTION, AL
		ret
	GET_OPTION2	ENDP

	;gets a string
	GET_STRING2	PROC
		;set max input length to 8 and GS_PRESS_BACK to 0 (usually menus will have 0 to
		;redirect the user to the previous page)
		mov		MAX_UNB_STR_LEN, 8
		mov		GS_PRESS_BACK, 0
		_GS2_ATTEMPT_GET_STRING:
		call	GET_UNBOUND_STRING

		;compare to see if the input length is 1 in case the user typed 0
		movzx	BX, _ACT_UNB_STR_LEN
		cmp		BX, 1
		
		;if string length > 1, user did not type '0'
		jne		_GS2_IS_NOT_BACK
		
		;then we check if the character really is '0'
		cmp		UNBOUND_STR_ARR[BX - 1], '0'
		
		;if not, the input is too short
		jne		_GS2_REATTEMPT_STR_INPUT_SHORT
		
		;if it is, set GS_PRESS_BACK to 1 and return
		mov		GS_PRESS_BACK, 1
		ret

		_GS2_IS_NOT_BACK:
		;check if the string is too short, reattempt if yes
		movzx	AX, MIN_UNB_STR_LEN
		cmp		BX, AX
		jle		_GS2_REATTEMPT_STR_INPUT_SHORT

		;loads all required variables before starting the continuous check
		mov		AL, _ACT_UNB_STR_LEN
		mov		IIC_INP_ARR_LEN, AL
		mov		IIC_INP_ARR_OFF, OFFSET UNBOUND_STR_ARR
		call	IS_INPUT_CONTINUOUS
		cmp		RES_IS_CONT, 0
		je		_GS2_REATTEMPT_STR_INPUT_NC
		
		;everything is valid, we stop the input
		jmp		_GS2_STOP_STR_INPUT

		;if the input is too short, we reprompt
		_GS2_REATTEMPT_STR_INPUT_SHORT:
		;clear input field
		movzx	CX, MAX_UNB_STR_LEN
		mov		_CL_INP_FIELD_LENGTH, CX
		call	_CLEAR_INPUT_FIELD

		;clear the error message
		mov		_ERROR_MSG_LEN, _NOTCONT_INPUT_LEN
		mov		AL, MAX_UNB_STR_LEN
		mov		_INP_FIELD_LEN, AL
		call	_CLEAR_ERROR_MSG
		;display error msg
		mov		_ERROR_MSG_OFFSET, OFFSET _FEW_NUM_INPUT_MSG
		mov		AL, MAX_UNB_STR_LEN
		mov		_ERROR_MSG_POS_OFFSET, AL
		call	_DISPLAY_ERROR_MSG
		jmp		_GS2_ATTEMPT_GET_STRING

		;if the string is not continuous (has spaces), we reprompt
		_GS2_REATTEMPT_STR_INPUT_NC:
		;clear input field
		movzx	CX, MAX_UNB_STR_LEN
		mov		_CL_INP_FIELD_LENGTH, CX
		call	_CLEAR_INPUT_FIELD

		;clear the error message
		mov		_ERROR_MSG_LEN, _NOTCONT_INPUT_LEN
		mov		AL, MAX_UNB_STR_LEN
		mov		_INP_FIELD_LEN, AL
		call	_CLEAR_ERROR_MSG

		;display error msg
		mov		_ERROR_MSG_OFFSET, OFFSET _NOTCONT_INPUT_MSG
		mov		AL, MAX_UNB_STR_LEN
		mov		_ERROR_MSG_POS_OFFSET, AL
		call	_DISPLAY_ERROR_MSG
		jmp		_GS2_ATTEMPT_GET_STRING

	_GS2_STOP_STR_INPUT:
		;clear the error message anyway
		mov		_ERROR_MSG_LEN, _NOTCONT_INPUT_LEN
		mov		AL, MAX_UNB_STR_LEN
		mov		_INP_FIELD_LEN, AL
		call	_CLEAR_ERROR_MSG
		ret
	GET_STRING2	ENDP

	;gets the seat according to the [row:col] input given by user
	;returns GOT_SEAT (index of seat in numbers)
	GET_SEAT	PROC
		;reset all variables required for this input method
		mov		GOT_SEAT, 0
		mov		MAX_UNB_STR_LEN, 2
		mov		DO_ECHO, 1
		
		_GS_ATTEMPT_GET_SEAT:
		call	GET_UNBOUND_STRING
		
		;compare if the input length is 2, as well as whether
		;the first character is '1' - '2' and the second character is 'A' - 'J'
		;(small letters are also taken into account)
		cmp		_ACT_UNB_STR_LEN, 2
		jnz		_GS_INVALID_SEAT_TYPE
		mov		BX, 0
		cmp		UNBOUND_STR_ARR[BX], '1'
		jl		_GS_INVALID_SEAT_TYPE
		cmp		UNBOUND_STR_ARR[BX], '2'
		jg		_GS_INVALID_SEAT_TYPE
		mov		BX, 1
		mov		AL, UNBOUND_STR_ARR[BX]
		cmp		AL, 'A'
		jl		_GS_INVALID_SEAT_TYPE
		cmp		AL, 'j'
		jg		_GS_INVALID_SEAT_TYPE
		
		;additional check for lower letters, since the ascii table has a gap between
		;'J' and 'a'
		cmp		AL, 'J'
		jg		_GS_IS_OOR_CAPITAL_LOWER
		jmp		_GS_DONE
		
		_GS_IS_OOR_CAPITAL_LOWER:
		cmp		AL, 'a'
		jl		_GS_INVALID_SEAT_TYPE
		jmp		_GS_DONE
		
		;if it's a valid id, check if the seat has already been taken
		;we remap the coordinate into a number and compare the value at ALL_SEATS[OFFSET] with 'X'
		;if it is 'X', it is taken and we have to ask again
		_GS_DONE:
		call	_CONVERT_TO_SEAT_ID
		mov		AL, SEAT_ID
		mov		SELECTED_SEAT_IDX, AL
		call	GET_SEAT_OFFSET
		mov		SI, SEAT_OFFSET
		cmp		ALL_SEATS[SI], 'X'
		je		_GS_SEAT_TAKEN
		
		;or it could also be a seat that the user has previously inputted, so we need to check that
		mov		AL, SELECTED_SEAT_IDX
		mov		TO_FIND, AL
		mov		INP_FIELD_OFFSET, OFFSET SELECTED_SEATS
		mov		ACT_FIELD_LEN, 6
		
		;find the seat id in the selected seats array
		call	FIND_IN
		cmp		ITEM_FOUND, 1
		je		_GS_SEAT_TAKEN
		jmp		_GS_SEAT_SUCCESS
		
		;seat is taken, we reprompt
		_GS_SEAT_TAKEN:
		; clear input field
		movzx	CX, MAX_UNB_STR_LEN
		mov		_CL_INP_FIELD_LENGTH, CX
		call	_CLEAR_INPUT_FIELD

		;clear the error message
		mov		_ERROR_MSG_LEN, INVALID_SEAT_TYPE_LEN
		mov		AL, MAX_UNB_STR_LEN
		mov		_INP_FIELD_LEN, AL
		call	_CLEAR_ERROR_MSG

		;display error msg
		mov		_ERROR_MSG_OFFSET, OFFSET SEAT_ALREADY_TAKEN_MSG
		mov		AL, MAX_UNB_STR_LEN
		mov		_ERROR_MSG_POS_OFFSET, AL
		call	_DISPLAY_ERROR_MSG
		jmp		_GS_ATTEMPT_GET_SEAT

		;invalid seat type, we reprompt
		_GS_INVALID_SEAT_TYPE:
		;clear input field
		movzx	CX, MAX_UNB_STR_LEN
		mov		_CL_INP_FIELD_LENGTH, CX
		call	_CLEAR_INPUT_FIELD

		;clear the error message
		mov		_ERROR_MSG_LEN, INVALID_SEAT_TYPE_LEN
		mov		AL, MAX_UNB_STR_LEN
		mov		_INP_FIELD_LEN, AL
		call	_CLEAR_ERROR_MSG

		;display error msg
		mov		_ERROR_MSG_OFFSET, OFFSET INVALID_SEAT_TYPE_MSG
		mov		AL, MAX_UNB_STR_LEN
		mov		_ERROR_MSG_POS_OFFSET, AL
		call	_DISPLAY_ERROR_MSG
		jmp		_GS_ATTEMPT_GET_SEAT
		
		;if successful, move the seat id into GOT_SEAT
		_GS_SEAT_SUCCESS:
		mov		GOT_SEAT, AL
		ret
	GET_SEAT	ENDP

	;gets a y/n response from the user
	;returns YN_RESPONSE (1 for yes, 0 for no)
	GET_YN	PROC
		;we want to echo the response, reset YN_RESPONSE to 0 and only accept 1 character
		mov		YN_RESPONSE, 0
		mov		MAX_UNB_STR_LEN, 1
		mov		DO_ECHO, 1

		_GYN_ATTEMPT_GET_YN:
		call	GET_UNBOUND_STRING
		
		;compare with all options ('Y', 'y', 'N', 'n')
		mov		BX, 0
		cmp		UNBOUND_STR_ARR[BX], 'y'
		je		_GYN_PROCEED_Y
		cmp		UNBOUND_STR_ARR[BX], 'Y'
		je		_GYN_PROCEED_Y
		cmp		UNBOUND_STR_ARR[BX], 'n'
		je		_GYN_PROCEED_N
		cmp		UNBOUND_STR_ARR[BX], 'N'
		je		_GYN_PROCEED_N
		jmp		_GYN_ERROR
		
		;none of the input was Y/N, so we tell them it is an invalid response
		_GYN_ERROR:
		;clear the input field
		movzx	CX, MAX_UNB_STR_LEN
		mov		_CL_INP_FIELD_LENGTH, CX
		call	_CLEAR_INPUT_FIELD

		;clear the error message
		mov		_ERROR_MSG_LEN, INVALID_YN_OPTION_LEN
		mov		AL, MAX_UNB_STR_LEN
		mov		_INP_FIELD_LEN, AL
		call	_CLEAR_ERROR_MSG

		;display the error message and reattempt the input
		mov		_ERROR_MSG_OFFSET, OFFSET INVALID_YN_OPTION_MSG
		mov		AL, MAX_UNB_STR_LEN
		mov		_ERROR_MSG_POS_OFFSET, AL
		call	_DISPLAY_ERROR_MSG
		jmp		_GYN_ATTEMPT_GET_YN

		;if we do however get a Y/y response, we set YN_RESPONSE to 1
		_GYN_PROCEED_Y:
		mov		YN_RESPONSE, 1
		
		_GYN_PROCEED_N:
		ret
	GET_YN	ENDP

	;gets a string in an array without <enter> taking an additional space,
	;as well as able to echo out the input/hide the input with an echo mask (e.g. '*')
	;requires DO_ECHO (whether the input should be echoed),
	;ECHO_MASK (in case DO_ECHO is 0, all characters input will be replaced with the mask)
	;MAX_UNB_STR_LEN (maximum 16) (the size of the input buffer)
	;returns UNBOUND_STR_ARR (the array of characters)
	GET_UNBOUND_STRING	PROC
		;let's make our own 0AH instruction but with optional echo/no echo and echo masks
		;reset _ACT_UNB_STR_LEN to 0 and clear the array by resetting all values to '$'
		mov		_ACT_UNB_STR_LEN, 0

		_GUS_CLEAR_STRING_INPUT_FIELD:
		mov		CX, _MAX_UNB_STR_LEN
		mov		SI, 0000

		;loop to clear all elements of the data
		_GUS_LCLSTR:
		mov		UNBOUND_STR_ARR[SI], '$'
		inc		SI
		loop	_GUS_LCLSTR

		;we get each char separately with 07H so that our input doesn't echo automatically
		;set the cursor position correctly before attempting input
		_GUS_GET_CHAR:
		push	CURSOR_POS
		movzx	BX, _ACT_UNB_STR_LEN
		add		CURSOR_POS, BX
		call	MOVE_CURSOR
		pop		CURSOR_POS
		mov		AH, 07H
		int		21H

		;check what type of character the user has entered
		cmp		AL, 08H
		je		_GUS_PRESSED_BACKSPACE
		cmp		AL, 0DH
		je		_GUS_PRESSED_ENTER

		;if no special action, check if we have filled up the buffer
		inc		_ACT_UNB_STR_LEN
		mov		CL, MAX_UNB_STR_LEN
		cmp		_ACT_UNB_STR_LEN, CL
		ja		_GUS_CHAR_OVERFLOW
		push	AX

		;input is not echoed, we put mask it with ECHO_MASK (default '*' for passwords)
		cmp		DO_ECHO, 1
		je		_GUS_ECHO
		mov		AH, 02H
		mov		DL, ECHO_MASK
		int		21H
		jmp		_GUS_ADD_TO_ARR

		;input is allowed to be echoed, we print the character out
		_GUS_ECHO:
		mov		AH, 02H
		mov		DL, AL
		int		21H
		jmp		_GUS_ADD_TO_ARR
		
		;then we add the character to the UNBOUND_STR_ARR array
		;then we repeat to get the next char
		_GUS_ADD_TO_ARR:
		pop		AX
		movzx	BX, _ACT_UNB_STR_LEN
		mov		UNBOUND_STR_ARR[BX - 1], AL
		jmp		_GUS_GET_CHAR

		;in case buffer is full, decrease back _ACT_UNB_STR_LEN to clamp it to the limit
		;then we repeat to get the next char
		_GUS_CHAR_OVERFLOW:
		dec		_ACT_UNB_STR_LEN
		jmp		_GUS_GET_CHAR

		;in case a backspace is pressed
		_GUS_PRESSED_BACKSPACE:
		dec		_ACT_UNB_STR_LEN
		cmp		_ACT_UNB_STR_LEN, 0
		jl		_GUS_CHAR_UNDERFLOW

		;reposition the cursor
		movzx	BX, _ACT_UNB_STR_LEN
		push	CURSOR_POS
		add		CURSOR_POS, BX
		call	MOVE_CURSOR
		pop		CURSOR_POS
		
		movzx	BX, _ACT_UNB_STR_LEN

		;fill the removed character with space to make it look responsive
		;then we reset the deleted character with '$'
		mov		AH, 02H
		mov		DL, ' '
		int		21H
		mov		UNBOUND_STR_ARR[BX], '$'
		jmp		_GUS_GET_CHAR
		
		;if we pressed enter, the procedure ends, input stops
		_GUS_PRESSED_ENTER:
		ret

		;underflow happens (backspace but the buffer is empty)
		;so we clamp _ACT_UNB_STR_LEN to 0 and continue getting a char
		_GUS_CHAR_UNDERFLOW:
		inc		_ACT_UNB_STR_LEN
		jmp		_GUS_GET_CHAR
	GET_UNBOUND_STRING	ENDP

;Misc-----------------------------------------------------------------------------------------------

	;copies the text from a buffer to another
	;requires ACT_FIELD_LEN (the length to copy),
	;INP_FIELD_OFFSET (the offset of the buffer to copy from)
	;DEST_FIELD_OFFSET (the offset of the buffer to copy to)
	COPY_TEXT	PROC
		;compare the length to copy
		cmp		ACT_FIELD_LEN, 0
		ja		_CT_START_COPY_TEXT
		
		;length is 0, doesn't require copying
		ret

		_CT_START_COPY_TEXT:
		;loads the input buffer to SI and the destination buffer to DI
		;sets CX to the length to copy
		mov		CX, ACT_FIELD_LEN
		mov		SI, INP_FIELD_OFFSET
		mov		DI, DEST_FIELD_OFFSET
		jmp		_CT_CONTINUE_COPY_TEXT

		;CX > 0, we continue copying
		_CT_CONTINUE_COPY_TEXT:
		;transfers the content of [SI] to [DI]
		;increases the index by 1
		mov		AL, [SI]
		mov		[DI], AL
		inc		SI
		inc		DI
		loop	_CT_CONTINUE_COPY_TEXT

		;text has been copied
		ret
	COPY_TEXT	ENDP

	;converts an array of characters into a 16-bit number
	;requires _ACT_FIELD_LEN (the length of the array) (used by get input procedures),
	;_FIELD_OFFSET (the offset of the array)
	;returns DEC_NUM (the result in number form)
	CONVERT_TO_HEX	PROC
		;compare the length of the array, and convert if length > 0
		cmp		_ACT_FIELD_LEN, 0
		ja		_CTH_START_MULT_DEC
		ret

		;we reset the result to 0 and power to 1
		;move the length of the array to CX for counting
		;move the offset of the array to SI for indexing
		_CTH_START_MULT_DEC:
		mov		_DEC_NUM, 0
		mov		_POWER, 1
		movzx	CX, _ACT_FIELD_LEN
		movzx	SI, _ACT_FIELD_LEN
		add		SI, _FIELD_OFFSET
		dec		SI
		jmp		_CTH_MULT_DEC

		;move AL to each individual digit in reverse order and sub the ASCII_CONSTANT
		;then multiply by BX (_POWER (10^n)) (16-bit multiplication)
		;then adds the result to the total (_DEC_NUM)
		_CTH_MULT_DEC:
		mov		AL, [SI]
		sub		AL, ASCII_CONST
		movzx	AX, AL
		mov		BX, _POWER
		mul		BX
		add		_DEC_NUM, AX

		loop	_CTH_CONTINUE_MULT_DEC
		jmp 	_CTH_EXIT_MULT_DEC

		;CX > 0, continue multiplication (8-bit multiplication)
		;we multiply BL by 10 and store the next (10^n) back to _POWER
		_CTH_CONTINUE_MULT_DEC:
		dec		SI
		mov		AX, _POWER
		mov		BL, 10
		mul		BL
		mov		_POWER, AX
		jmp		_CTH_MULT_DEC

		;CX = 0, and we are done converting
		;copy _DEC_NUM to DEC_NUM to avoid overwriting
		_CTH_EXIT_MULT_DEC:
		mov		AX, _DEC_NUM
		mov		DEC_NUM, AX
		ret
	CONVERT_TO_HEX	ENDP

	;converts a hex number (16-bit) into an array of digits
	;requires HEX_NUM (hex number variable)
	_CONVERT_TO_DECIMAL	PROC
		cmp		HEX_NUM, 0
		jae		_CTD_START_DIV_HEX

		;input is <= 0, error might occur
		ret

		;hex number >= 0, let's start converting it
		;and reset the _SPC_INDEX to 0
		_CTD_START_DIV_HEX:
		mov		CX, 5
		mov		SI, 0
		_CTD_LCLSTR:
		mov		_DEC_NM_RES[SI], '$'	; loop to clear all elements of the data
		inc		SI
		loop	_CTD_LCLSTR

		mov		AX, HEX_NUM
		mov		BX, 10
		mov		_SPC_INDEX, 0
		jmp 	_CTD_DIV_HEX

		;clear DX and move divisor (10) to BX and divide by 10
		_CTD_DIV_HEX:
		xor		DX, DX
		div		BX
		cmp		DL, 0

		;continue loop if DL != 0
		ja		_CTD_CONTINUE_DIV_HEX
		
		;final exit check
		jz		_CTD_IS_CONTINUE_DIV_HEX

		;remainder(DL) > 0 (remainder will always be a byte long)
		;so we continue dividing
		;replace the next item of _DEC_NM_RES with the remainder (reverse order)
		_CTD_CONTINUE_DIV_HEX:
		movzx	SI, _SPC_INDEX
		mov		_DEC_NM_RES[SI], DL
		inc 	_SPC_INDEX
		jmp		_CTD_DIV_HEX

		;if DL is 0, it doesn't neccessarily mean AX has been fully divided
		;as it may contain the final digit, so we make one last check
		_CTD_IS_CONTINUE_DIV_HEX:
		cmp		AX, 0

		;continue loop if AX > 0
		jnz		_CTD_CONTINUE_DIV_HEX
		
		;AX is 0, but has AX been 0 at the time of procedure call?
		cmp		_SPC_INDEX, 0
		je		_CTD_HEX_NUM_IS_ZERO
		jmp		_CTD_EXIT_DIV_HEX

		;AX was 0 when this procedure was called (_SPC_INDEX was 0, no division occured)
		;so we just assign a digit 0 at the beginning
		_CTD_HEX_NUM_IS_ZERO:
		movzx	SI, _SPC_INDEX
		mov		_DEC_NM_RES[SI], 0
		inc		_SPC_INDEX
		jmp		_CTD_EXIT_DIV_HEX

		;we are done, decrease _SPC_INDEX to adjust to the correct index for printing
		_CTD_EXIT_DIV_HEX:
		dec		_SPC_INDEX
		ret
	_CONVERT_TO_DECIMAL ENDP

	;prints the decimal equivalent of a hex number (16-bit) onto the screen
	;requires HEX_NUM (hex number variable)
	PRINT_DECIMAL	PROC
		;HEX_NUM is already passed, _CONVERT_TO_DECIMAL will also use it
		call	_CONVERT_TO_DECIMAL

		;hex number has been converter, let's display it
		;AH = 02H -> display character instruction
		mov		AH, 02H
		jmp		_PD_PRINT_NUM
		
		;we load the current integer in reverse order (since _SPC_INDEX is decrementing, it will be correct)
		;BX = _SPC_INDEX (the index of the current digit) (modified by CONVERT_TO_HEX)
		;AL = _DEC_NM_RES[BX] (the current digit)
		;we add the ASCII_CONSTANT to convert to its character counterpart
		;DL = the converted char
		_PD_PRINT_NUM:
		movzx	BX, _SPC_INDEX
		mov		AL, _DEC_NM_RES[BX]
		add		AL, ASCII_CONST
		mov		DL, AL
		int 	21H

		cmp		_SPC_INDEX, 0
		
		;_SPC_INDEX is still > 0, continue printing
		jnz		_PD_CONTINUE_PRINT_NUM
		
		;we're done with all the numbers
		je		_PD_EXIT_PRINT_DECIMAL

		;decrease the _SPC_INDEX, and jump back to printing
		_PD_CONTINUE_PRINT_NUM:
		dec		_SPC_INDEX
		jmp		_PD_PRINT_NUM

		;exit
		_PD_EXIT_PRINT_DECIMAL:
		ret
	PRINT_DECIMAL	ENDP

	;gets the smallest value of the two numbers
	;requires MIN_1 (first number to compare),
	;MIN_2 (second number to compare),
	;returns MINIMUM (the minimum between the 2 numbers)
	GET_MINIMUM		PROC
		mov		AH, MIN_1
		mov		AL, MIN_2
		cmp		AH, AL
		jb		_MIN_AH_LESSER
		mov		MINIMUM, AL
		ret
		
		_MIN_AH_LESSER:
		mov		MINIMUM, AH
		ret
	GET_MINIMUM		ENDP
